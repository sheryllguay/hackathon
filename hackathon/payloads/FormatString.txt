# Usage: pass as the user-controlled string that printf() echoes back verbatim.
# Goal: recover a code pointer (PIE base + offset) or a libc pointer (0x7f..).

%p %p %p %p %p %p          # dump first N words; find 0x55../0x56.. code or 0x7f.. libc
%llx                       # unsigned long long hex (full 64-bit on x86-64)
